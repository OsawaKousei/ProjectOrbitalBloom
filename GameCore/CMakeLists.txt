cmake_minimum_required(VERSION 3.16)
project(OrbitalBloomGameCore LANGUAGES CXX)
add_library(orbital_core src/Simulation.cpp)
target_include_directories(orbital_core PUBLIC include)
target_compile_features(orbital_core PUBLIC cxx_std_17)
enable_testing()
add_executable(orbital_core_tests tests/SimulationTests.cpp)
target_link_libraries(orbital_core_tests PRIVATE orbital_core)
add_test(NAME orbital_core_tests COMMAND orbital_core_tests)
